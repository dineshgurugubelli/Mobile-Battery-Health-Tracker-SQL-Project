mysql> CREATE TABLE battery_usage (
    ->     usage_id INT PRIMARY KEY AUTO_INCREMENT,
    ->     device_id INT NOT NULL,
    ->     usage_date DATE NOT NULL,
    ->     battery_start INT NOT NULL,
    ->     battery_end INT NOT NULL,
    ->     screen_time_hours DECIMAL(4,2),
    ->     battery_consumed INT,
    ->
    ->     FOREIGN KEY (device_id) REFERENCES mobile_devices(device_id),
    ->
    ->     CHECK (battery_start BETWEEN 0 AND 100),
    ->     CHECK (battery_end BETWEEN 0 AND 100),
    ->     CHECK (battery_end <= battery_start),
    ->     CHECK (screen_time_hours >= 0)
    -> );
Query OK, 0 rows affected (0.03 sec)

Question:
--->Which mobile has the best overall battery performance based on battery health, charging cycles, battery consumption, and reported issues?

mysql> SELECT
    ->     m.brand,
    ->     m.model,
    ->     bh.health_percentage,
    ->     bh.charging_cycles,
    ->     SUM(bu.battery_consumed) AS total_battery_consumed,
    ->     COUNT(DISTINCT bi.issue_id) AS total_battery_issues,
    ->
    ->     CASE
    ->         WHEN bh.health_percentage >= 90
    ->              AND bh.charging_cycles < 200
    ->              AND COUNT(DISTINCT bi.issue_id) = 0
    ->             THEN 'Excellent'
    ->
    ->         WHEN bh.health_percentage >= 80
    ->              AND bh.charging_cycles < 300
    ->             THEN 'Good'
    ->
    ->         WHEN bh.health_percentage >= 70
    ->             THEN 'Average'
    ->
    ->         ELSE 'Poor'
    ->     END AS overall_battery_performance
    ->
    -> FROM mobile_devices m
    ->
    -> JOIN battery_health bh
    ->     ON m.device_id = bh.device_id
    ->
    -> JOIN battery_usage bu
    ->     ON m.device_id = bu.device_id
    ->
    -> LEFT JOIN battery_issues bi
    ->     ON m.device_id = bi.device_id
    ->
    -> GROUP BY
    ->     m.device_id,
    ->     m.brand,
    ->     m.model,
    ->     bh.health_percentage,
    ->     bh.charging_cycles
    ->
    -> ORDER BY
    ->     bh.health_percentage DESC,
    ->     total_battery_issues ASC;
